-- ══════════════════════════════════════
--  CRIPTO (port exacto del Worker: lane/digest, SALT "rgh4|k|")
-- ══════════════════════════════════════
local SALT = "rgh4|k|"
local B32 = (type(bit32) == "table") and bit32 or nil

local function bxor(a, b)
    a = math.floor(a) % 4294967296
    b = math.floor(b) % 4294967296
    if B32 then return B32.bxor(a, b) end
    local r, p = 0, 1
    for _ = 1, 32 do
        local aa, bb = a % 2, b % 2
        if aa ~= bb then r = r + p end
        a = (a - aa) / 2
        b = (b - bb) / 2
        p = p * 2
    end
    return r
end
local function rshift(a, n) return math.floor(math.floor(a) / 2 ^ n) end
local function imul(a, b)   -- multiplicación 32-bit exacta sin bitops
    local ah, al = math.floor(a / 65536), math.floor(a) % 65536
    local bh, bl = math.floor(b / 65536), math.floor(b) % 65536
    return (al * bl + ((ah * bl + al * bh) % 65536) * 65536) % 4294967296
end
local function hex8(n) return string.format("%08x", n) end
local function lane(s, seed)
    local h = seed % 4294967296
    for i = 1, #s do
        h = bxor(h, string.byte(s, i))
        h = imul(h, 16777619)
        h = rshift(h, 15)
    end
    h = bxor(h, #s)
    h = imul(bxor(h, rshift(h, 16)), 0x85ebca6b)
    h = imul(bxor(h, rshift(h, 13)), 0xc2b2ae35)
    return bxor(h, rshift(h, 16))
end
local function digestFn(s)
    return hex8(lane(s, 0x811c9dc5)) .. hex8(lane(s .. "\0" .. SALT, 0x9e3779b9))
end
digest = digestFn

-- Decodificador XOR + Base64 (sin bitops: compatible con cualquier ejecutor)
local ENC_B64 = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local function encXor(a, b)
    local r, p = 0, 1
    for _ = 1, 8 do
        local aa, bb = a % 2, b % 2
        if aa ~= bb then r = r + p end
        a, b, p = (a - aa) / 2, (b - bb) / 2, p * 2
    end
    return r
end
local function encB64Decode(s)
    s = tostring(s or ""):gsub("[^" .. ENC_B64 .. "=]", "")
    local bits, n = {}, 0
    s:gsub(".", function(c)
        if c ~= "=" then
            local f = ENC_B64:find(c, 1, true) - 1
            for i = 6, 1, -1 do
                n = n + 1
                bits[n] = (f % 2 ^ i - f % 2 ^ (i - 1) > 0) and "1" or "0"
            end
        end
    end)
    local out = {}
    for i = 1, n - 7, 8 do
        local v = 0
        for j = 0, 7 do
            if bits[i + j] == "1" then v = v + 2 ^ (7 - j) end
        end
        out[#out + 1] = string.char(math.floor(v))
    end
    return table.concat(out)
end
local function encDecode(t)
    if type(t) ~= "table" or type(t.key) ~= "string" or type(t.data) ~= "string" or t.key == "" then return "" end
    local raw = encB64Decode(t.data)
    if raw == "" then return "" end
    local k, out = t.key, {}
    for i = 1, #raw do
        out[#out + 1] = string.char(encXor(raw:byte(i), k:byte((i - 1) % #k + 1)))
    end
    return table.concat(out)
end

    local t = THEMES[name] or THEMES["Verde Neón"]
    for k, v in pairs(t) do M[k] = v end
    for _, r in ipairs(ThemeReg) do
        pcall(function()
            if r.kind == "grad" then

            end
        end)

    for _, fn in ipairs(ThemeFns) do pcall(fn) end -- V4.1: refresca elementos animados
    return name
end

local function makeDraggable(obj, allowStart)
    local dragging, moved = false, false
    local ds, sp = nil, nil
    addConn(obj.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch
        or i.UserInputType == Enum.UserInputType.MouseButton1 then
            if allowStart and not allowStart(i) then return end
            dragging = true
            moved = false
            ds = Vector2.new(i.Position.X, i.Position.Y)
            sp = obj.Position
        end
    end))
    addConn(UIS.InputChanged:Connect(function(i)
        if not dragging then return end
        if i.UserInputType == Enum.UserInputType.Touch
        or i.UserInputType == Enum.UserInputType.MouseMovement then
            -- V4.1: compensa la escala de UI para que el dedo siga 1:1 el elemento
            local s = math.max(UIScaleObj.Scale, 0.05)
            local d = (Vector2.new(i.Position.X, i.Position.Y) - ds) / s
            if d.Magnitude > 8 then moved = true end
            if moved then
                obj.Position = UDim2.new(sp.X.Scale, sp.X.Offset + d.X, sp.Y.Scale, sp.Y.Offset + d.Y)
            end
        end
    end))
    addConn(UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch
        or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end))
    return function() return moved end
end

        if draggingSlider and (i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseMovement) then

-- ══════════════════════════════════════
--  GUARD — cliente del Cloudflare Worker
--  Nunca viajan secretos desde el cliente: solo IDs públicos + hash de key.
-- ══════════════════════════════════════
local Guard = {}
Guard.session = string.format("%08x%08x", math.random(0, 2147483647), math.random(0, 2147483647))
Guard.lastT = 0
Guard.lastRes = nil

local function guardURL(path)
    return (CONFIG.WorkerURL:gsub("/+$", "")) .. path
end

function Guard.verifySig(x, u, banned, keyvalid, role, s)
    if not WORKER_SECRET or WORKER_SECRET == "" then return true end -- sin secreto configurado no se valida
    local expect = digest(x .. "|" .. u .. "|" .. (banned and "1" or "0") .. "|" ..
        (keyvalid and "1" or "0") .. "|" .. tostring(role) .. "|" .. WORKER_SECRET)
    return string.lower(tostring(s or "")) == expect
end

function Guard.check(keyHash, force)
    local now = os.clock()
    if not force and Guard.lastRes and (now - Guard.lastT) < 3 then
        return Guard.lastRes
    end
    Guard.lastT = now
    if not CONFIG.WorkerURL:find("https://", 1, true) or CONFIG.WorkerURL:find("TU%-SUBDOMINIO", 1, false) then
        return nil, "WorkerURL sin configurar"
    end
    local t0 = os.clock()
    local q = "?u=" .. tostring(lp.UserId) ..
        "&n=" .. HttpSvc:UrlEncode(lp.Name:lower()) ..
        "&x=" .. Guard.session ..
        "&h=" .. HttpSvc:UrlEncode(getHwid()) ..
        "&f=" .. (force and "1" or "0")
    if keyHash then q = q .. "&k=" .. keyHash end

    local res = nil
    for attempt = 1, (CONFIG.RequestRetry + 1) do
        local r = httpReq({ Url = guardURL("/check" .. q), Method = "GET" })
        if r and r.StatusCode == 200 and r.Body then
            local body = jdec(r.Body)
            if body and body.ok then res = body; break end
        end
        if attempt <= CONFIG.RequestRetry then task.wait(0.6) end
    end

    if res then
        res.latency = math.floor((os.clock() - t0) * 1000)
        res.sigOK = Guard.verifySig(Guard.session, tostring(lp.UserId),
            res.banned == true, res.keyvalid == true, res.role, res.s)
        res.t = os.time()
        Guard.lastRes = res
        Auth.lastRes = res
        return res
    end
    return nil, "sin respuesta del Guard"
end

function Guard.admin(path, body)
    if not Auth.ownerKey or Auth.ownerKey == "" then return nil, "sin Owner Key (ingrésala arriba)" end
    local r = httpReq({
        Url = guardURL(path),
        Method = "POST",
        Headers = {
            ["content-type"] = "application/json",
            ["X-Owner-Id"] = tostring(lp.UserId),
            ["X-Owner-Key"] = Auth.ownerKey,
        },
        Body = jenc(body or {}),
    })
    if r and r.Body then
        local b = jdec(r.Body)
        if b then
            if b.error then
                return nil, tostring(b.error) .. (b.why and (" (" .. tostring(b.why) .. ")") or "")
            end
            return b
        end
    end
    return nil, "sin respuesta del Worker"
end

local function getRoot()
    local c = lp.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end
local function getHum()
    local c = lp.Character
    return c and c:FindFirstChildOfClass("Humanoid")
end
local function tpTo(part)
    local r = getRoot()
    if r and part then
        pcall(function() r.CFrame = part.CFrame + Vector3.new(0, 4, 0) end)
    end
end
local function getRooms()
    local rf = workspace:FindFirstChild("Rooms")
    if not rf then return {} end
    local t = {}
    for _, r in ipairs(rf:GetChildren()) do
        if tonumber(r.Name) then table.insert(t, { n = tonumber(r.Name), model = r }) end
    end
    table.sort(t, function(a, b) return a.n < b.n end)
    return t
end
local function tpSafe()
    local rooms = workspace:FindFirstChild("Rooms")
    if not rooms then return false end
    for _, r in ipairs(rooms:GetChildren()) do
        local n = r.Name:lower()
        if n:find("safe") or n:find("clock") or n:find("church") then
            local p = r:FindFirstChildWhichIsA("BasePart", true)
            if p then tpTo(p); return true end
        end
    end
    return false
end

local function setFullbright(on)
    state.fullbright = on
    if on then
        pcall(function() Lighting.Ambient = Color3.new(1, 1, 1) end)
        pcall(function() Lighting.Brightness = 3 end)
        pcall(function() Lighting.GlobalShadows = false end)
        pcall(function() Lighting.Technology = Enum.Technology.Compatibility end)
        pcall(function() Lighting.FogEnd = 9e9 end)
        pcall(function() Lighting.OutdoorAmbient = Color3.new(1, 1, 1) end)
        for _, v in ipairs(Lighting:GetChildren()) do
            if v:IsA("PostProcessEffect") or v:IsA("Atmosphere") then
                pcall(function() v.Enabled = false end)
            end
        end
        Notify("Fullbright ON", M.GOLD)
    else
        for k, v in pairs(origLight) do pcall(function() Lighting[k] = v end) end
        for v, st in pairs(origEffectStates) do pcall(function() v.Enabled = st end) end
        Notify("Fullbright OFF", M.GRAY)
    end
end

-- Anti-Lag / FPS Booster (con throttle real y restauración)
local antiLagBtnRef = nil
local function applyAntiLag(enable)
    state.antiLag = enable
    pcall(function() settings().Network.IncomingReplicationLag = 0 end)
    pcall(function()
        settings().Physics.PhysicsEnvironmentalThrottle = enable
            and Enum.EnviromentalPhysicsThrottle.Disabled
            or Enum.EnviromentalPhysicsThrottle.DefaultAuto
    end)
    pcall(function() Lighting.GlobalShadows = not enable end)
    if enable then
        task.spawn(function()
            local count = 0
            for _, obj in ipairs(workspace:GetDescendants()) do
                if not state.antiLag then break end
                count = count + 1
                pcall(function()
                    if obj:IsA("BasePart") then
                        obj.Material = Enum.Material.SmoothPlastic
                        obj.CastShadow = false
                    elseif obj:IsA("Decal") or obj:IsA("Texture") then
                        obj.Transparency = 1
                    elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") then
                        obj.Enabled = false
                    end
                end)
                if count % 80 == 0 then task.wait() end
            end
        end)
        Notify("FPS Booster ON", M.ACCENT)
        if antiLagBtnRef then
            antiLagBtnRef.Text = "  ⚡ FPS Booster — ON"
            antiLagBtnRef.TextColor3 = M.ACCENT
        end
    else
        Notify("FPS Booster OFF", M.GRAY)
        if antiLagBtnRef then
            antiLagBtnRef.Text = "  ⚡ FPS Booster — OFF"
            antiLagBtnRef.TextColor3 = M.GRAYLT
        end
    end
end

local function buildFlyPad()
    destroyFlyPad()
    local pad = Instance.new("Frame", SG)
    pad.Size = UDim2.new(0, 150, 0, 100)
    pad.Position = UDim2.new(1, -165, 1, -120)
    pad.BackgroundTransparency = 1
    pad.ZIndex = 250
    table.insert(flyPadParts, pad)
    local defs = {
        { k = "f", t = "▲", x = 0.5, y = 0 },
        { k = "b", t = "▼", x = 0.5, y = 0.66 },
        { k = "l", t = "◀", x = 0,   y = 0.33 },
        { k = "r", t = "▶", x = 0.66, y = 0.33 },
        { k = "u", t = "⬆", x = 1,   y = 0 },
        { k = "d", t = "⬇", x = 1,   y = 0.66 },
    }
    for _, d in ipairs(defs) do
        local b = Instance.new("TextButton", pad)
        b.Size = UDim2.new(0, 48, 0, 46)
        b.Position = UDim2.new(d.x, d.x >= 1 and -48 or (d.x >= 0.5 and -24 or 0), d.y, 0)
        b.BackgroundColor3 = M.BG2
        b.BorderSizePixel = 0
        b.Font = Enum.Font.GothamBold
        b.Text = d.t
        b.TextColor3 = M.ACCENT
        b.TextSize = 16
        b.ZIndex = 251
        Corner(b, 10)
        Stroke(b, nil, 1, "ACCENTD")
        b.InputBegan:Connect(function() flyDirs[d.k] = true end)
        b.InputEnded:Connect(function() flyDirs[d.k] = false end)
        table.insert(flyPadParts, b)
    end
end

local function setFly(on)
    state.fly = on
    if on then
        local r = getRoot()
        if not r then
            Notify("Sin personaje para activar Fly", M.RED)
            state.fly = false
            return
        end
        if flyBV then pcall(function() flyBV:Destroy() end) end
        flyBV = Instance.new("BodyVelocity")
        flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        flyBV.Velocity = Vector3.new(0, 0, 0)
        flyBV.Parent = r
        local h = getHum()
        if h then h.PlatformStand = true end
        flyConn = addConn(RunService.RenderStepped:Connect(function()
            if not flyBV or not flyBV.Parent then return end
            local c = cam()
            if not c then return end
            local cf = c.CFrame
            local v = Vector3.new(0, 0, 0)
            if UIS:IsKeyDown(Enum.KeyCode.W) or flyDirs.f then v = v + cf.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.S) or flyDirs.b then v = v - cf.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.A) or flyDirs.l then v = v - cf.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.D) or flyDirs.r then v = v + cf.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.Space) or flyDirs.u then v = v + Vector3.new(0, 1, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftShift) or flyDirs.d then v = v - Vector3.new(0, 1, 0) end
            if v.Magnitude > 0 then
                flyBV.Velocity = v.Unit * state.flySpeed
            else
                flyBV.Velocity = Vector3.new(0, 0, 0)
            end
        end))
        if UIS.TouchEnabled then buildFlyPad() end
        Notify("Fly ON" .. (UIS.TouchEnabled and " — usa el pad táctil" or " — WASD + Space/Shift"), M.ACCENT)
    else
        if flyConn then flyConn:Disconnect(); flyConn = nil end
        if flyBV then pcall(function() flyBV:Destroy() end); flyBV = nil end
        destroyFlyPad()
        local h = getHum()
        if h then h.PlatformStand = false end
        Notify("Fly OFF", M.GRAY)
    end
end

local function loadConfig()
    local s = fileRead(CONFIG.ConfigFile)
    local t = s and jdec(s)
    if type(t) == "table" then
        if t.theme and THEMES[t.theme] then cfg.theme = t.theme end
        if type(t.scale) == "number" then cfg.scale = math.clamp(t.scale, 0.75, 1.35) end
        if type(t.notifDur) == "number" then cfg.notifDur = math.clamp(t.notifDur, 2, 8) end
        if type(t.fpsHud) == "boolean" then cfg.fpsHud = t.fpsHud end
    end
    NotifDur = cfg.notifDur
end
local function saveConfig()
    fileWrite(CONFIG.ConfigFile, jenc(cfg))
end

    table.sort(themeNames)

        for j = i, math.min(i + 1, #themeNames) do

            table.insert(pair, { label = "🎨 " .. nm, cb = function()
                if nm == "Oro VIP" and not hasPerm("vip") then

        end
        mkBtnRow(T, pair)

    end

        cfg.notifDur = v

        saveConfig()
    end, 0)

    mkSection(T, "💾 Sistema")
    mkBtnRow(T, {
        { label = "💾 Guardar ahora", cb = function()
            saveConfig()
            Notify("Ajustes guardados", M.ACCENT, 2.5)

local function buildEstado(T)
    mkSection(T, "📡 Estado del Guard")
    estadoRows.estado = mkValueRow(T, "Conexión", function() return "-" end, false)
    estadoRows.lat    = mkValueRow(T, "Latencia", function() return "-" end, false)
    estadoRows.rol    = mkValueRow(T, "Rol", function() return "-" end, false)
    estadoRows.key    = mkValueRow(T, "Key", function() return "-" end, false)
    estadoRows.sig    = mkValueRow(T, "Firma", function() return "-" end, false)

    estadoRows.mem    = mkValueRow(T, "En Discord", function() return "-" end, false)

    estadoRows.hwid   = mkValueRow(T, "Huella HWID", function() return getHwid() end, true)

    local function refreshEstado()
        local res = Auth.lastRes
        local set = function(r, txt) if r and r.set then pcall(r.set, txt) end end
        set(estadoRows.estado, Auth.status == "ONLINE" and "GUARD en línea ✓" or "SIN GUARD (offline)")
        set(estadoRows.lat, res and (tostring(res.latency) .. " ms") or "-")
        set(estadoRows.rol, Auth.role)
        set(estadoRows.key, res and (res.keyvalid and "válida ✓" or "no válida ✗") or "-")
        set(estadoRows.sig, res and (res.sigOK and "firma OK ✓" or "FIRMA INVÁLIDA ✗") or "-")
        set(estadoRows.flag, res and (res.flag and "⚠ posible reventa" or "ok") or "-")
        set(estadoRows.mem, res and (res.unknown and "¿? (sin datos)" or (res.member and "sí ✓" or "no ✗")) or "-")
        set(estadoRows.last, res and os.date("%d/%m %H:%M:%S", res.t) or "-")
    end
    _G.RGH_V4.RefreshEstado = refreshEstado

    mkSection(T, "🔐 Firma y secret")
    if WORKER_SECRET == "" then
        mkCard(T, "⚠ Sin SECRET configurado",
            "No hay secreto en el script: las respuestas del Guard no se validan. Genera tu bloque con: node rg_tools.js encode \"TU-SECRETO\" y pégalo en CONFIG.SecretEnc.", M.ORANGE)

    mkSection(T, "🖥 Servidor actual")

            task.wait(0.05)

                Notify("Guard OK — rol " .. Auth.role, M.ACCENT, 3)

                .. "Rol: " .. Auth.role .. "\n"

        "Acceso: OWNER / CO-OWNER / DEVELOPER. Todas las acciones se validan en el Worker con tu UserId + Owner Key. La Owner Key NUNCA se escribe en el script.", M.PURPLE)

    mkSection(T, "🔑 Owner Key")
    local okIn = mkInput(T, "pega tu OWNER_KEY (queda en memoria y archivo local)")
    if Auth.ownerKey then okIn.Text = Auth.ownerKey end
    mkBtnRow(T, {
        { label = "💾 Usar / recordar", cb = function()
            local v = okIn.Text:gsub("%s", "")
            if v == "" then Notify("Escribe la Owner Key", M.RED, 3) return end
            Auth.ownerKey = v
            fileWrite(CONFIG.OwnerFile, v)
            Notify("Owner Key lista (memoria + archivo local)", M.ACCENT, 3)
        end },
        { label = "🧪 Probar", cb = function()
            if not Auth.ownerKey or Auth.ownerKey == "" then Notify("Sin Owner Key", M.RED, 3) return end
            Notify("Consultando /diag...", M.CYAN, 2)
            task.wait(0.05)
            local d = Guard.diag()
            if d and d.owner == "ok" then
                Notify("Owner autenticada en el Guard ✓", M.ACCENT, 4)
            elseif d then
                Notify("Guard dice: " .. tostring(d.owner), M.RED, 4)
            else
                Notify("Sin respuesta del Worker", M.RED, 4)
            end
        end },
    })

    mkSection(T, "⛔ Baneos globales")
    local banIn = mkInput(T, "id:123456 · name:usuario · hwid:xxx · key:hash (o escribe tal cual)")
    local motivoIn = mkInput(T, "motivo del baneo (opcional)")
    mkBtnRow(T, {
        { label = "⛔ Ban ID", cb = function()
            local v = tostring(banIn.Text:gsub("^id:", ""))
            local r, e = Guard.admin("/ban", { t = "id", v = v, reason = motivoIn.Text, by = lp.Name })
            Notify(r and ("Baneado: " .. tostring(r.banned)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 4)
        end },
        { label = "⛔ Ban Nombre", cb = function()
            local v = tostring(banIn.Text:gsub("^name:", ""))
            local r, e = Guard.admin("/ban", { t = "name", v = v, reason = motivoIn.Text, by = lp.Name })
            Notify(r and ("Baneado: " .. tostring(r.banned)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 4)
        end },
    })
    mkBtnRow(T, {
        { label = "⛔ Ban key (hash)", cb = function()
            local v = tostring(banIn.Text:gsub("^key:", "")):lower()
            local r, e = Guard.admin("/ban", { t = "key", v = v, reason = motivoIn.Text, by = lp.Name })
            Notify(r and ("Key baneada: " .. tostring(r.banned)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 4)
        end },
        { label = "✅ Unban (auto)", cb = function()
            local raw = tostring(banIn.Text):gsub("%s", "")
            if raw == "" then Notify("Escribe el valor (o tipo:valor)", M.GOLD, 3) return end
            local t, v = raw:match("^(%a+):(.+)$")
            if not t then
                if raw:match("^%d+$") then t, v = "id", raw else t, v = "name", raw end
            end
            t = t:lower()
            if t == "user" or t == "usuario" then t = "name" end
            if not (({ id = true, name = true, hwid = true, key = true })[t]) then
                Notify("Tipo desconocido: " .. t, M.RED, 3) return
            end
            local r, e = Guard.admin("/unban", { t = t, v = v, by = lp.Name })
            Notify(r and ("Desbaneado: " .. tostring(r.unbanned)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 4)
        end },
    })
    mkBtnRow(T, {
        { label = "📋 Listar baneos", cb = function()
            local r, e = Guard.admin("/list", {})
            if r and r.bans then
                local out = {}
                for _, b in ipairs(r.bans) do out[#out + 1] = b.k .. " — " .. tostring(b.reason or "") end
                clipboard(table.concat(out, "\n"))
                Notify(#out .. " baneos copiados al portapapeles", M.ACCENT, 3)
            else
                Notify("Error: " .. tostring(e), M.RED, 4)
            end
        end },
        { label = "✅ Unban HWID", cb = function()
            local raw = tostring(banIn.Text):gsub("%s", ""):gsub("^hwid:", "")
            if raw == "" then Notify("Escribe la huella HWID", M.GOLD, 3) return end
            local r, e = Guard.admin("/unban", { t = "hwid", v = raw, by = lp.Name })
            Notify(r and "HWID desbaneado" or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 4)
        end },
    })

    -- 🧹 V4.1: revierte un baneo que se extendió a más usuarios (2 toques = confirmación)
    local purgeBtn
    local purgeArmed = false
    purgeBtn = mkBtn(T, "🧹 Desbanear TODOS (2 toques)", Color3.fromRGB(60, 16, 16), function()
        if not purgeArmed then
            purgeArmed = true
            if purgeBtn then purgeBtn.Text = "  ⚠ TOCA OTRA VEZ PARA CONFIRMAR" end
            task.delay(6, function()
                purgeArmed = false
                if purgeBtn and purgeBtn.Parent then purgeBtn.Text = "  🧹 Desbanear TODOS (2 toques)" end
            end)
            return
        end
        purgeArmed = false
        if purgeBtn then purgeBtn.Text = "  ⏳ Limpiando baneos..." end
        task.spawn(function()
            local r, e = Guard.admin("/list", {})
            if not (r and r.bans) then
                Notify("Error: " .. tostring(e), M.RED, 4)
                if purgeBtn and purgeBtn.Parent then purgeBtn.Text = "  🧹 Desbanear TODOS (2 toques)" end
                return
            end
            local n = 0
            for _, bn in ipairs(r.bans) do
                local t, v = tostring(bn.k or ""):match("^ban:([^:]+):(.+)$")
                if t and v then
                    if Guard.admin("/unban", { t = t, v = v, by = lp.Name }) then n = n + 1 end
                    task.wait(0.12)
                end
                if n >= 150 then break end
            end
            if purgeBtn and purgeBtn.Parent then purgeBtn.Text = "  🧹 Desbanear TODOS (2 toques)" end
            Notify("Limpieza lista — " .. n .. " baneos eliminados. Vuelve a banear SOLO al UserId correcto.", M.ACCENT, 6)
        end)
    end)

    mkSection(T, "🛡 Roles (override en servidor)")
    local roleIn = mkInput(T, "UserId para asignar rol")
    mkBtnRow(T, {
        { label = "OWNER", cb = function()
            local r, e = Guard.admin("/role", { v = roleIn.Text, role = "OWNER", by = lp.Name })
            Notify(r and ("Rol asignado: " .. tostring(r.role)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 3)
        end },
        { label = "CO-OWNER", cb = function()
            local r, e = Guard.admin("/role", { v = roleIn.Text, role = "CO-OWNER", by = lp.Name })
            Notify(r and ("Rol asignado: " .. tostring(r.role)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 3)
        end },
        { label = "DEVELOPER", cb = function()
            local r, e = Guard.admin("/role", { v = roleIn.Text, role = "DEVELOPER", by = lp.Name })
            Notify(r and ("Rol asignado: " .. tostring(r.role)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 3)
        end },
    })
    mkBtnRow(T, {
        { label = "ADMIN", cb = function()
            local r, e = Guard.admin("/role", { v = roleIn.Text, role = "ADMIN", by = lp.Name })
            Notify(r and ("Rol asignado: " .. tostring(r.role)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 3)
        end },
        { label = "MODERATOR", cb = function()
            local r, e = Guard.admin("/role", { v = roleIn.Text, role = "MODERATOR", by = lp.Name })
            Notify(r and ("Rol asignado: " .. tostring(r.role)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 3)
        end },
        { label = "PREMIUM", cb = function()
            local r, e = Guard.admin("/role", { v = roleIn.Text, role = "PREMIUM", by = lp.Name })
            Notify(r and ("Rol asignado: " .. tostring(r.role)) or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 3)
        end },
    })
    mkBtnRow(T, {
        { label = "USER (quita override)", cb = function()
            local r, e = Guard.admin("/role", { v = roleIn.Text, role = "USER", by = lp.Name })
            Notify(r and "Override de rol quitado" or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 3)
        end },
    })

    mkSection(T, "🔑 Keys del Guard")
    local keyIn = mkInput(T, "key EN TEXTO (solo esta vez: se guarda como hash)")
    local noteIn = mkInput(T, "nota de la key (opcional)")
    mkBtnRow(T, {
        { label = "➕ Registrar key", cb = function()
            local r, e = Guard.admin("/keyadd", { k = keyIn.Text, note = noteIn.Text, by = lp.Name })
            if r then
                Notify("Key registrada — hash: " .. tostring(r.hash), M.ACCENT, 5)
                keyIn.Text = ""
            else
                Notify("Error: " .. tostring(e), M.RED, 4)
            end
        end },
        { label = "➖ Borrar key", cb = function()
            local r, e = Guard.admin("/keydel", { k = keyIn.Text, by = lp.Name })
            Notify(r and "Key borrada del Guard" or ("Error: " .. tostring(e)), r and M.ACCENT or M.RED, 3)
        end },
        { label = "📋 Listar (hashes)", cb = function()
            local r, e = Guard.admin("/keylist", {})
            if r and r.keys then
                local out = {}
                for _, k in ipairs(r.keys) do out[#out + 1] = k.hash .. " — " .. tostring(k.note or "") end
                clipboard(table.concat(out, "\n"))
                Notify(#out .. " keys copiadas (solo hashes)", M.ACCENT, 3)
            else
                Notify("Error: " .. tostring(e), M.RED, 4)
            end
        end },
    })

    mkSection(T, "📊 Diagnóstico y logs")
    mkBtnRow(T, {
        { label = "👥 Perfiles recientes", cb = function()
            local r, e = Guard.admin("/profiles", {})
            if r and r.profiles then
                local out = {}
                for _, p in ipairs(r.profiles) do
                    out[#out + 1] = p.id .. " @" .. tostring(p.n or "?") .. " (" .. tostring(p.e or "?") .. ")"
                end
                clipboard(table.concat(out, "\n"))
                Notify(#out .. " perfiles copiados", M.ACCENT, 3)
            else
                Notify("Error: " .. tostring(e), M.RED, 4)
            end
        end },
        { label = "📨 Test de log", cb = function()
            Guard.log("Prueba de log V4",
                "Enviado desde el hub por @" .. lp.Name,
                { { name = "UserId", value = tostring(lp.UserId), inline = true } },
                0x00e64d)
            Notify("Log enviado al Worker (requiere WEBHOOK configurado)", M.ACCENT, 4)
        end },
    })
end

-- ══════════════════════════════════════
--  ARRANQUE AUTENTICADO
-- ══════════════════════════════════════
local function boot()
    task.spawn(function()
        setLoad(0.08, "Cargando ajustes...")
        loadConfig()
        applyTheme(cfg.theme)
        UIScaleObj.Scale = cfg.scale
        NotifDur = cfg.notifDur
        task.wait(0.15)

        setLoad(0.18, "Conectando con ROSHAN GUARD...")
        local res, err = Guard.check(Auth.keyHash, false)

        if res and res.banned then
            banScreen(res.reason)
            if Auth.banned or not SG.Parent then return end
        end

        local online = false
        if res then
            online = true
            Auth.status = "ONLINE"
            Auth.role = res.role or "USER"
            Auth.lastRes = res
            if res.sigOK == false then
                Notify("⚠ Respuesta del Guard con firma inválida — revisa el SECRET", M.ORANGE, 7, "⚠")
            end
        else
            Auth.status = "OFFLINE"
            Notify("Guard sin conexión (" .. tostring(err) .. ") — modo offline", M.ORANGE, 6, "⚠")
            if CONFIG.StrictMode then
                setLoad(0.2, "Modo ESTRICTO: sin Guard no hay acceso.")
                task.wait(4)
                if _G.RGH_V4 and _G.RGH_V4.Destroy then _G.RGH_V4.Destroy() end
                return
            end
            if Auth.keyHash and not LOCAL_KEYS[Auth.keyHash] then
                Auth.keyHash = nil
            end
        end

        local needsKey = false
        if online then
            needsKey = (res.keyvalid == false)
        else
            needsKey = (Auth.keyHash == nil)
        end
        if needsKey then
            keyScreen(online)
            if Auth.banned or not SG.Parent then return end
        end

        setLoad(0.45, "Autenticado — rol: " .. Auth.role)
        task.wait(0.1)
        setLoad(0.55, "Construyendo interfaz...")

        registerTabs()
        buildTabsUI()
        for i, def in ipairs(tabDefs) do
            pcall(def.build, i)
        end

        setLoad(0.85, "Aplicando ajustes...")
        setFpsHud(cfg.fpsHud)
        updateRoleUI()
        pcall(_G.RGH_V4.RefreshEstado)

        setLoad(0.93, "Sincronizando perfil con el Guard...")
        Guard.sendProfile()
        task.wait(0.2)

        loaderDone()
        MF.Visible = true
        shadow.Visible = true
        hubVisible = true
        RBtn.Visible = true

        Notify("Bienvenido " .. lp.DisplayName .. " — rol " .. Auth.role, M.ACCENT, 5, "★")
        task.wait(0.5)
        Notify("Tecla [R] o botón R para abrir/cerrar", M.ACCENT2, 5, "ℹ")
        if WORKER_SECRET == "" then
            Notify("Sin SECRET: respuestas sin validar (pestaña Estado)", M.ORANGE, 7, "⚠")
        elseif digest(WORKER_SECRET):sub(1, 8) == DEFAULT_SECRET_FP then
            Notify("⚠ SECRET de fábrica — cámbialo (pestaña Estado)", M.ORANGE, 8, "⚠")
        end
    end)
end

-- ══════════════════════════════════════
--  API PÚBLICA + DESTROY (limpieza total, sin fugas)
-- ══════════════════════════════════════
_G.RGH_V4.Version = CONFIG.Version
_G.RGH_V4.Role = function() return Auth.role end
_G.RGH_V4.Notify = Notify

_G.RGH_V4.Destroy = function()
    -- 1. apagar todos los estados
    for k in pairs(state) do
        if type(state[k]) == "boolean" then state[k] = false end
    end
    Auth.banned = false

    -- 2. desconectar TODAS las conexiones registradas
    for _, c in ipairs(Conns) do
        pcall(function() c:Disconnect() end)
    end
    Conns = {}

    -- 3. limpiar ESP y atributos
    espHLList = clearHL(espHLList)
    pespHLList = clearHL(pespHLList)
    bespHLList = clearHL(bespHLList)
    pcall(function()
        for _, obj in ipairs(workspace:GetChildren()) do
            if obj.GetAttribute and obj:GetAttribute("RGH_ESP") then
                obj:SetAttribute("RGH_ESP", nil)
            end
        end
        for _, pl in ipairs(Players:GetPlayers()) do
            local c = pl.Character
            if c and c:GetAttribute("RGH_ESP") then c:SetAttribute("RGH_ESP", nil) end
        end
        for _, pl in ipairs(Players:GetPlayers()) do
            local c = pl.Character
            local h = c and c:FindFirstChild("HumanoidRootPart")
            local tag = h and h:FindFirstChild("RGH_PlayerTag")
            if tag then tag:Destroy() end
        end
    end)

    -- 4. restaurar personaje
    pcall(function()
        local h = getHum()
        if h then
            h.PlatformStand = false
            pcall(function() h.WalkSpeed = 16 end)
        end
        local c = lp.Character
        if c then
            for _, p in ipairs(c:GetDescendants()) do
                if p:IsA("BasePart") then pcall(function() p.CanCollide = true end) end
            end
        end
    end)

    -- 5. restaurar cámara e iluminación
    if camBackup then
        pcall(function()
            lp.CameraMaxZoomDistance = camBackup.maxz
            lp.CameraMinZoomDistance = camBackup.minz
            lp.CameraMode = camBackup.mode
        end)
    end
    pcall(function()
        for k, v in pairs(origLight) do Lighting[k] = v end
        for v, st in pairs(origEffectStates) do v.Enabled = st end
    end)
    pcall(function()
        settings().Physics.PhysicsEnvironmentalThrottle = Enum.EnviromentalPhysicsThrottle.DefaultAuto
        settings().Network.IncomingReplicationLag = 0
    end)

    -- 6. destruir UI (panel + capa de overlays)
    pcall(function() SG:Destroy() end)
    pcall(function() SGO:Destroy() end)
    _G.RGH_V4 = nil
end

-- ══════════════════════════════════════
--  STARTUP
-- ══════════════════════════════════════
boot()
